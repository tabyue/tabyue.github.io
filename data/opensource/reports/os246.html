def rollout_in_world_model(wm, policy, s0, T):
    hist = wm.init_history(s0)
    for t in range(T):
        a = policy.act(hist.latest_obs())
        obs = wm.step(hist, a)          # 动作条件化的下一观测
        hist.append(obs)
        if wm.drift_detector(hist) > 0.5:
            return hist, 'aborted: drift'   # 长时程漂移，评测作废
    return hist, 'ok'
