# Fixed-base vs free-floating base: the same policy, two dynamics regimes.
def compare_regimes(policy, make_env, seeds=(0, 1, 2)):
    out = {}
    for regime in ('fixed_base', 'free_floating'):
        rets = []
        for s in seeds:
            env = make_env(regime=regime, seed=s)
            rets.append(policy.evaluate(env))
        out[regime] = {'mean': sum(rets) / len(rets), 'all': rets}
    out['regime_gap'] = out['fixed_base']['mean'] - out['free_floating']['mean']
    return out

  1. 域差可测性：能否方便切换固定基座与自由漂浮基座，这是它相对普通 Isaac 环境最核心的价值；

  4. 与主流 RL 栈的兼容性：能否不改动策略代码就接入现有训练脚本。
