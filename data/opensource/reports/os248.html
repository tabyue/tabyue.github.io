def loop(perception, policy, controller, sim):
    obs = perception.read(sim)             # 感知
    action = policy.act(obs)               # 决策（RL / WAM / VLA 之一）
    cmd = controller.track(action)         # 控制
    sim.step(cmd)                          # 执行并产生下一帧观测
    return obs, action, cmd
